```typescript
interface ReadinessMetrics {
  // F2.2 Core
  overallScore: number;
  confidenceScore: number;
  confidenceLevel: 'High' | 'Medium' | 'Low' | 'Insufficient Data';
  estimatedExamDayScore: number;
  trend: 'improving' | 'stable' | 'declining' | 'unknown';
  isCalibrated: boolean;
  
  // F2.2 Breakdown (9 dimensions)
  breakdown: {
    mockExams: number;        // out of 30
    subjectMastery: number;   // out of 25
    memoryRetention: number;  // out of 15
    practiceQuizzes: number;  // out of 10
    consistency: number;      // out of 8
    reviewCompletion: number; // out of 5
    learningVelocity: number; // out of 3
    studyTime: number;        // out of 2
    confidence: number;       // out of 2
  };
  
  // F2.2.5 Three Dimensions
  dimensions: {
    knowledge: number;  // 0-100
    memory: number;     // 0-100
    exam: number;       // 0-100
  };
  
  // F2.2.5 Potential Engine
  potential: {
    currentReadiness: number;
    potentialReadiness: number;
    potentialGain: number;
    recommendations: ReadinessRecommendation[];
    opportunityAnalysis: {
      lostToOverdueReviews: number;
      lostToWeakRetention: number;
      lostToInconsistency: number;
      lostToLowMockExposure: number;
      lostToUnfinishedTopics: number;
    };
  };
  
  // Metadata
  subjectReadiness: Record<string, number>;
  rawMetrics: {
    mockExamsTaken: number;
    totalQuestionsAnswered: number;
    daysActive: number;
  };
}
```

## Technical Requirements

### Data Fetching Strategy
**Recommended:** Refactor dashboard to use `DashboardDataService`

**Before:**
```tsx
// Current: Direct queries in dashboard page
const readinessMetrics = await calculateReadiness(user.id);
const { data: mockExams } = await supabase.from("mock_exam_attempts")...
// ... 10+ more queries
```

**After:**
```tsx
// Proposed: Use centralized service
const { data: dashboardData } = await getDashboardData(user.id);
// All data available in one object
```

**Benefits:**
- ✅ Single source of truth
- ✅ Consolidated error handling
- ✅ Cached data
- ✅ Parallel fetching
- ✅ Graceful degradation
